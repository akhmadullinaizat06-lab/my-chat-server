class DataManager:
    def __init__(self):
        self.base_salary = 1095.6
        self.tax_percent = 13.0
        self.is_dark = False
        self.history = []
        self.counts = [0] * 7
        self.load_all()

    def load_all(self):
        if os.path.exists("settings.json"):
            try:
                with open("settings.json", 'r') as f:
                    data = json.load(f)
                    self.is_dark = data.get('dark_mode', False)
                    self.base_salary = data.get('base_salary', 1095.6)
                    self.tax_percent = data.get('tax_percent', 13.0)
            except: pass
        if os.path.exists("history.json"):
            try:
                with open("history.json", 'r', encoding='utf-8') as f:
                    self.history = json.load(f)
            except: pass
        if os.path.exists("draft.json"):
            try:
                with open("draft.json", 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    if len(data.get("counts", [])) == 7:
                        self.counts = data["counts"]
            except: pass

    def save_settings(self):
        with open("settings.json", 'w') as f:
            json.dump({'dark_mode': self.is_dark, 'base_salary': self.base_salary, 'tax_percent': self.tax_percent}, f)

    def save_history(self):
        with open("history.json", 'w', encoding='utf-8') as f:
            json.dump(self.history, f, ensure_ascii=False, indent=4)

    def save_draft(self):
        with open("draft.json", 'w', encoding='utf-8') as f:
            json.dump({"counts": self.counts}, f)

    def clear_all(self):
        self.history = []
        self.counts = [0] * 7
        self.base_salary = 1095.6
        self.tax_percent = 13.0
        self.save_history()
        self.save_draft()
        self.save_settings()

class ViewModel:
    def __init__(self, data_manager):
        self.data = data_manager
        self.rates = [3.0, 4.0, 5.0, 6.0, 6.0, 6.0, 7.0]
    
    def calculate_totals(self):
        total_count = sum(self.data.counts)
        total_earned = sum(c * m for c, m in zip(self.data.counts, self.rates))
        gross = total_earned + self.data.base_salary
        tax = gross * (self.data.tax_percent / 100)
        net = gross - tax
        return {
            "total_count": total_count,
            "total_earned": round(total_earned, 2),
            "tax": round(tax, 2),
            "net": round(net, 2)
        }
    
    def get_dashboard_data(self):
        p1_total = 0; p2_total = 0; p1_days = 0; p2_days = 0
        for record in self.data.history:
            if record.get("period") == "25":
                p1_total += record['net']; p1_days += 1
            else:
                p2_total += record['net']; p2_days += 1
        return {
            "p1_total": round(p1_total, 0), "p2_total": round(p2_total, 0),
            "p1_days": p1_days, "p2_days": p2_days
        }

class Card(BoxLayout):
    def __init__(self, bg_color=(1,1,1,1), radius=20, **kwargs):
        super().__init__(**kwargs)
        self.padding = dp(15)
        self.radius = radius
        self.bg_color = bg_color
        with self.canvas.before:
            Color(0, 0, 0, 0.04)
            self.shadow = RoundedRectangle(pos=(self.x, self.y - dp(3)), size=(self.width, self.height), radius=[dp(self.radius)], smooth=True)
        with self.canvas.before:
            self.color_instr = Color(*self.bg_color)
            self.rect = RoundedRectangle(pos=self.pos, size=self.size, radius=[dp(self.radius)], smooth=True)
        self.bind(pos=self.update_rect, size=self.update_rect)

    def set_bg(self, color):
        self.color_instr.rgba = color

    def update_rect(self, *args):
        self.rect.pos = self.pos
        self.rect.size = self.size
        self.shadow.pos = (self.x, self.y - dp(3))
        self.shadow.size = (self.width, self.height)

class CounterRow(BoxLayout):
    count = NumericProperty(0)
    def __init__(self, title, subtitle, money_per_pick, on_change_callback, index, **kwargs):
        super().__init__(**kwargs)
        self.orientation = 'horizontal'
        self.size_hint_y = None
        self.height = dp(80)
        self.spacing = dp(10)
        self.money_per_pick = money_per_pick
        self.on_change = on_change_callback
        self.index = index
        
        left_box = BoxLayout(orientation='vertical', size_hint_x=0.45)
        self.lbl_title = Label(text=title, bold=True, font_size=sp(15), halign="left", valign="bottom")
        self.lbl_title.bind(size=self.lbl_title.setter('text_size'))
        self.lbl_sub = Label(text=subtitle, font_size=sp(12), halign="left", valign="top")
        self.lbl_sub.bind(size=self.lbl_sub.setter('text_size'))
        left_box.add_widget(self.lbl_title)
        left_box.add_widget(self.lbl_sub)
        
        self.plate = Card(radius=10, size_hint=(None, None), size=(dp(130), dp(42)), padding=0)
        self.plate.pos_hint = {"center_y": 0.5}
        
        self.btn_minus = Button(text="-", background_normal='', background_color=(0,0,0,0), font_size=sp(22))
        self.btn_minus.bind(on_press=self.on_btn_press, on_release=self.decrease)
        self.lbl_count = Label(text="0", bold=True, font_size=sp(18), size_hint_x=0.4)
        self.btn_plus = Button(text="+", background_normal='', background_color=(0,0,0,0), font_size=sp(22))
        self.btn_plus.bind(on_press=self.on_btn_press, on_release=self.increase)
        
        self.plate.add_widget(self.btn_minus)
        self.plate.add_widget(self.lbl_count)
        self.plate.add_widget(self.btn_plus)
        
        self.lbl_money = Label(text="0 ₽", bold=True, font_size=sp(15), halign="right", size_hint_x=0.3)
        self.lbl_money.bind(size=self.lbl_money.setter('text_size'))

        self.add_widget(left_box)
        self.add_widget(self.plate)
        self.add_widget(self.lbl_money)

    def on_btn_press(self, instance):
        Animation(color=(0.6, 0.6, 0.6, 1), duration=0.05).start(instance)

    def apply_theme(self, t):
        self.lbl_title.color = t['TEXT_MAIN']
        self.lbl_sub.color = t['TEXT_SEC']
        self.lbl_count.color = t['TEXT_MAIN']
        self.btn_minus.color = t['TEXT_MAIN']
        self.btn_plus.color = t['TEXT_MAIN']
        self.lbl_money.color = t['GREEN']
        self.plate.set_bg(t['PLATE'])

    def update_money(self):
        money = self.count * self.money_per_pick
        self.lbl_money.text = f"{int(money) if money.is_integer() else round(money, 2)} ₽"
        if self.on_change:
            self.on_change(self.index, self.count)

    def increase(self, instance):
        Animation(color=App.get_running_app().get_theme()['TEXT_MAIN'], duration=0.1).start(instance)
        self.count += 1
        self.lbl_count.text = str(self.count)
        self.update_money()

    def decrease(self, instance):
        Animation(color=App.get_running_app().get_theme()['TEXT_MAIN'], duration=0.1).start(instance)
        if self.count > 0:
            self.count -= 1
            self.lbl_count.text = str(self.count)
            self.update_money()

        target_w = btn_w - dp(10)
        
        Animation(pos=(target_x, self.y), size=(target_w, self.height), duration=0.25, t='out_cubic').start(self.rect_highlight)

        self.color_bg.rgba = t['TAB_BG']

        self.size_hint = (1, 1)
        self.opacity = 0
        
        with self.canvas.before:
            self.bg_color = Color(0, 0, 0, 0.4)
            self.bg_rect = RoundedRectangle(pos=self.pos, size=self.size)
        self.bind(pos=self.update_bg, size=self.update_bg)
        
        self.panel_container = RelativeLayout(size_hint=(1, 1))
        self.add_widget(self.panel_container)
        
        with self.panel_container.canvas.before:
            self.translate = Translate(0, 0)
        
        self.panel = FloatLayout(size_hint=(0.8, 1), pos_hint={"x": 0, "top": 1})

        content = BoxLayout(orientation='vertical', padding=dp(20), spacing=dp(15),
                            size_hint=(1, 1), pos_hint={"x": 0, "y": 0})
        
        content.add_widget(Label(text="Настройки", font_size=sp(24), bold=True, size_hint_y=None, height=dp(50)))
        
        t = app.get_theme()
        
        self.input_sal = TextInput(text=str(app.data.base_salary), multiline=False, 
                                   background_color=t['PLATE'], foreground_color=t['TEXT_MAIN'], 
                                   padding=[dp(12), dp(12)], font_size=sp(18), size_hint_y=None, height=dp(50))
        self.input_tax = TextInput(text=str(app.data.tax_percent), multiline=False, 
                                   background_color=t['PLATE'], foreground_color=t['TEXT_MAIN'], 
                                   padding=[dp(12), dp(12)], font_size=sp(18), size_hint_y=None, height=dp(50))
        
        content.add_widget(Label(text="Оклад", color=t['TEXT_SEC'], halign="left", size_hint_y=None, height=dp(20)))
        content.add_widget(self.input_sal)
        content.add_widget(Label(text="Налог (%)", color=t['TEXT_SEC'], halign="left", size_hint_y=None, height=dp(20)))
        content.add_widget(self.input_tax)
        
        btn_clear = Button(text="Очистить все данные", background_normal='', background_color=(0,0,0,0), color=t['RED'], size_hint_y=None, height=dp(40))
        btn_clear.bind(on_release=self.confirm_clear)
        content.add_widget(btn_clear)
        
        btn_apply = Button(text="Применить", background_normal='', background_color=t['BLUE'], color=(1,1,1,1), size_hint_y=None, height=dp(50), bold=True)
        btn_apply.bind(on_release=self.apply)
        content.add_widget(btn_apply)
        
        btn_close = Button(text="Закрыть", background_normal='', background_color=(0,0,0,0), color=t['TEXT_SEC'], size_hint_y=None, height=dp(50))
        btn_close.bind(on_release=self.hide)
        content.add_widget(btn_close)
        
        self.panel.add_widget(content)
        
        self.is_open = False
        self.translate.x = -Window.width * 0.8
        
    def update_bg(self, *args):
        self.bg_rect.pos = self.pos
        self.bg_rect.size = self.size

    def update_panel_bg(self, *args):
        self.panel_rect.pos = self.panel.pos
        self.panel_rect.size = self.panel.size

    def open(self):
        if self.is_open: return
        self.is_open = True
        self.opacity = 1
        Animation(x=0, duration=0.3, t='out_cubic').start(self.translate)
        Animation(opacity=1, duration=0.2).start(self)
    
    def hide(self, instance):
        if not self.is_open: return
        self.is_open = False
        Animation(x=-Window.width * 0.8, duration=0.25, t='in_cubic').start(self.translate)
        anim = Animation(opacity=0, duration=0.25)
        anim.bind(on_complete=lambda *args: setattr(self, 'opacity', 0))
        anim.start(self)
    
    def apply(self, instance):
        try:
            self.app.data.base_salary = float(self.input_sal.text)
            self.app.data.tax_percent = float(self.input_tax.text)
            self.app.data.save_settings()
            self.app.update_ui()
            self.hide(instance)
        except ValueError:
            self.input_sal.background_color = (1, 0.8, 0.8, 1)
            self.input_tax.background_color = (1, 0.8, 0.8, 1)

    def confirm_clear(self, instance):
        if hasattr(self, 'cleared_once'):
            self.app.data.clear_all()
            self.app.update_ui()
            self.app.update_history_ui()
            self.app.update_dashboard()
            self.hide(instance)
        else:
            self.cleared_once = True
            instance.text = "Вы уверены? Нажмите ещё раз"
            instance.color = (1, 0.5, 0, 1)

class ErrorToast(RelativeLayout):
    def __init__(self, **kwargs):

            self.rect = RoundedRectangle(pos=self.pos, size=self.size, radius=[dp(20)])
        
        self.bind(pos=self.update_rect, size=self.update_rect)

            valign="middle"
        )
        self.add_widget(self.lbl)

        Animation(opacity=0, duration=0.3).start(self)
        Clock.schedule_once(lambda dt: App.get_running_app().root_layout.remove_widget(self), 0.3)

class SalaryApp(App):
    def build(self):
        self.data = DataManager()
        self.vm = ViewModel(self.data)
        self.root_layout = RelativeLayout()
        self.sm = ScreenManager()
        
        self.drawer = SettingsDrawer(self)
        
        self.build_calc_screen()
        self.build_stats_screen()
        self.build_history_screen()
        self.root_layout.add_widget(self.sm)
        
        self.tab_bar = GlassTabBar(self)
        self.root_layout.add_widget(self.tab_bar)
        
        self.root_layout.add_widget(self.drawer)
        
        # --- ДОБАВЛЯЕМ ОБРАБОТКУ СВАЙПОВ ---
        self.root_layout.bind(on_touch_down=self.on_swipe_start)
        self.root_layout.bind(on_touch_up=self.on_swipe_end)
        self.start_x = 0
        self.start_y = 0
        self.is_swiping = False
        
        self.apply_theme()
        self.switch_screen("calc")
        return self.root_layout

    def on_swipe_start(self, instance, touch):
        # Если панель открыта, свайпы работают иначе (для закрытия)
        if self.drawer.is_open:
            if touch.x < Window.width * 0.2:  # Начало у левого края
                self.start_x = touch.x
                self.start_y = touch.y
                self.is_swiping = True
            return False
        
        # Обычные свайпы (по экрану)
        if self.sm.collide_point(*touch.pos):
            self.start_x = touch.x
            self.start_y = touch.y
            self.is_swiping = True
        return False

    def on_swipe_end(self, instance, touch):
        if not self.is_swiping:
            return False
        self.is_swiping = False
        
        dx = touch.x - self.start_x
        dy = touch.y - self.start_y
        
        # Проверка на горизонтальный свайп
        if abs(dx) > abs(dy) and abs(dx) > dp(50):
            if self.drawer.is_open:
                # Если панель открыта и свайп вправо - закрываем
                if dx > 0:
                    self.drawer.hide(None)
                return False
            
            current = self.sm.current
            
            # Свайп влево
            if dx < 0:
                if current == "calc": self.switch_screen("stats")
                elif current == "stats": self.switch_screen("history")
                elif current == "history": self.drawer.open()
            # Свайп вправо
            elif dx > 0:
                if current == "stats": self.switch_screen("calc")
                elif current == "history": self.switch_screen("stats")
        return False

    def show_error(self):
        toast = ErrorToast()
        self.root_layout.add_widget(toast)

    def get_theme(self):
        return THEMES["dark"] if self.data.is_dark else THEMES["light"]

    def apply_theme(self):
        t = self.get_theme()
        Window.clearcolor = t['BG']
        
        self.lbl_calc_title.color = t['TEXT_MAIN']
        self.btn_reset.color = t['RED']
        self.work_card.set_bg(t['CARD'])
        self.work_title.color = t['TEXT_SEC']
        for c in self.counters: c.apply_theme(t)
        self.total_card.set_bg(t['PLATE'])
        self.lbl_total_count.color = t['TEXT_MAIN']
        self.lbl_total_earned.color = t['TEXT_MAIN']
        self.lbl_tax.color = t['RED']
        self.lbl_final.color = t['GREEN']
        self.btn_save.background_color = t['GREEN']
        
        self.lbl_stats_title.color = t['TEXT_MAIN']
        self.card1.set_bg(t['BLUE_LIGHT'])
        self.lbl_p1.color = t['BLUE']
        self.lbl_period1.color = t['BLUE']
        self.lbl_days1.color = t['BLUE']
        self.card2.set_bg(t['CARD'])
        self.lbl_p2.color = (0.7, 0.5, 0.2, 1)
        self.lbl_period2.color = (0.7, 0.5, 0.2, 1)
        self.lbl_days2.color = (0.7, 0.5, 0.2, 1)
        
        self.lbl_hist_title.color = t['TEXT_MAIN']
        self.update_history_ui()
        
        self.tab_bar.set_theme(t)

    def switch_theme(self, instance):
        self.data.is_dark = not self.data.is_dark
        self.data.save_settings()
        self.apply_theme()

    def build_calc_screen(self):
        t = self.get_theme()
        screen = Screen(name="calc")
        layout = BoxLayout(orientation='vertical')

    def switch_screen(self, name):
        self.sm.current = name
        self.tab_bar.set_theme(self.get_theme())
        self.tab_bar.update_highlight_position()
        if name == "stats": self.update_dashboard()
        if name == "history": self.update_history_ui()

    def update_count_from_ui(self, index, count):
        self.data.counts[index] = count
        self.update_ui()
        Clock.unschedule(self.save_draft_delayed)
        Clock.schedule_once(self.save_draft_delayed, 1.0)

    def save_draft_delayed(self, dt):
        self.data.save_draft()

    def update_ui(self):
        totals = self.vm.calculate_totals()
        self.lbl_total_count.text = f"{totals['total_count']} шт."
        self.lbl_total_earned.text = f"{totals['total_earned']} ₽"
        self.lbl_tax.text = f"{totals['tax']} ₽"
        self.lbl_final.text = f"{totals['net']} ₽"

    def reset_counters(self, instance):
        self.data.counts = [0] * 7
        for c in self.counters:
            c.count = 0
            c.lbl_count.text = "0"
            c.lbl_money.text = "0 ₽"
        self.update_ui()
        self.data.save_draft()

    def save_day(self, instance):
        total_count = sum(self.data.counts)
        if total_count == 0:
            return
        
        totals = self.vm.calculate_totals()
        net = totals['net']
        
        current_date = datetime.now().strftime("%d.%m")
        current_day = datetime.now().day
        period = "25" if current_day <= 15 else "10"
        
        record = {"date": current_date, "net": net, "period": period}
        self.data.history.append(record)
        self.data.save_history()
            
        self.reset_counters(None)
        self.switch_screen("history")

    def update_dashboard(self):
        dash = self.vm.get_dashboard_data()
        self.lbl_period1.text = f"{dash['p1_total']} ₽"
        self.lbl_period2.text = f"{dash['p2_total']} ₽"
        self.lbl_days1.text = f"Отработано дней: {dash['p1_days']}"
        self.lbl_days2.text = f"Отработано дней: {dash['p2_days']}"

    def update_history_ui(self):
        t = self.get_theme()
        self.hist_box.clear_widgets()
        if not self.data.history:
            self.hist_box.add_widget(Label(text="История пуста", color=t['TEXT_SEC'], size_hint_y=None, height=dp(50)))
            return
            
        for record in reversed(self.data.history):
            card = Card(radius=15, size_hint_y=None, height=dp(70), padding=dp(15))
            card.set_bg(t['CARD'])
            card.add_widget(Label(text=record['date'], color=t['TEXT_MAIN'], bold=True, size_hint_x=0.3, halign="left", font_size=sp(16)))
            period_text = "→ 25 числа" if record.get('period') == "25" else "→ 10 числа"
            card.add_widget(Label(text=period_text, color=t['TEXT_SEC'], size_hint_x=0.4, font_size=sp(14)))
            card.add_widget(Label(text=f"{record['net']} ₽", color=t['GREEN'], bold=True, size_hint_x=0.3, halign="right", font_size=sp(16)))
            
            btn_del = Button(text="✕", background_normal='', background_color=(0,0,0,0), color=t['TEXT_SEC'], size_hint_x=None, width=dp(40), font_size=sp(18))
            btn_del.bind(on_release=lambda x, r=record: self.delete_record(r))
            card.add_widget(btn_del)
            
            self.hist_box.add_widget(card)

    def delete_record(self, record):
        if record in self.data.history:
            self.data.history.remove(record)
            self.data.save_history()
            self.update_history_ui()
            self.update_dashboard()

    def open_settings(self, instance):
        self.drawer.open()
